cmake_minimum_required(VERSION 3.15)
project(FMLAxionHMcode LANGUAGES CXX)
find_package(GSL REQUIRED)
get_filename_component(FML_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_library(fml_axionhmcode AxionHMcode.cpp ../HaloModel/HaloModelTools.cpp)
target_compile_features(fml_axionhmcode PUBLIC cxx_std_17)
target_compile_definitions(fml_axionhmcode PUBLIC USE_GSL)
target_include_directories(fml_axionhmcode PUBLIC "${FML_INCLUDE}")
target_link_libraries(fml_axionhmcode PUBLIC GSL::gsl GSL::gslcblas)
add_executable(axionhmcode_example example/example.cpp)
target_link_libraries(axionhmcode_example PRIVATE fml_axionhmcode)
include(CTest)
if(BUILD_TESTING)
    add_executable(test_axionhmcode tests/test_axionhmcode.cpp "${CMAKE_CURRENT_SOURCE_DIR}/../Global/Global.cpp")
    target_compile_definitions(test_axionhmcode PRIVATE AXION_REFERENCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/reference")
    target_link_libraries(test_axionhmcode PRIVATE fml_axionhmcode)
    add_test(NAME axionhmcode COMMAND test_axionhmcode)
endif()
